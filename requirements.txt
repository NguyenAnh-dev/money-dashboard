# Requires Python 3.10+ -- the codebase uses `X | None` unions.
# Developed and tested against 3.13.
rich==15.0.0

# Plaid (Step 4.7). RUNTIME, but only on the `plaid` branch: data_source.py
# imports the adapter INSIDE the branch, so `DATA_SOURCE=fake` never loads any
# of this. That matters -- it is a large SDK with its own transitive deps
# (urllib3, python-dateutil, six, nulltype), and none of them should be
# imported to render a list of tuples.
#
# Pinned exactly, like everything else here. The v9+ API shape (Configuration /
# ApiClient / PlaidApi + one request MODEL class per endpoint) is completely
# different from the pre-9 `plaid.Client`, so any snippet found online has to
# be checked against this version before it means anything.
plaid-python==42.0.0

# Reads .env into os.environ so DATA_SOURCE / PLAID_* can be set per machine
# without exporting them by hand. Never commits: .env is gitignored, and
# .env.example is the committed record of which keys exist.
python-dotenv==1.2.2

# Password hashing (Step 6.2). RUNTIME, and deliberately NOT hashlib: a
# general-purpose hash is built to be fast, which is the one property a
# password hash must not have. argon2id is the current default recommendation
# and the library ships sane parameters plus a rehash signal.
#
# Pulls argon2-cffi-bindings, cffi and pycparser -- a C extension, so this is
# the first dependency here that is not pure Python. Nothing under adapters/
# imports it; core/passwords.py is the only module that does.
argon2-cffi==25.1.0

# Test-only. Kept in the same file rather than a requirements-dev.txt because
# there is one environment here (./venv) and one person in it; a split would be
# two files to forget to install. Nothing under core/, adapters/ or main.py may
# import pytest.
pytest==9.1.1
